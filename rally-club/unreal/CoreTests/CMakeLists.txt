cmake_minimum_required(VERSION 3.16)
project(rally_core_tests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(rally_core_tests CoreTests.cpp ../Source/RallyClub/Core/RallyCore.cpp ../Source/RallyClub/Core/RallyMatch.cpp)
enable_testing()
add_test(NAME rally_core_tests COMMAND rally_core_tests)
